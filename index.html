async function syncAllSheetData() { const toast = document.getElementById('status-toast'); toast.className = 'mb-6 p-4 rounded-xl text-xs font-medium flex items-center justify-between glass-card border-blue-500/40 text-blue-300'; toast.innerHTML = ` Streaming public CSV layers from Google Sheet ID...`; toast.classList.remove('hidden'); const tabs = { attendance: 'Attendance', performance: 'Performance Data', regionwise: 'Regionwise Data', tat: 'TAT', quality: 'Quality' }; for (const [key, tabName] of Object.entries(tabs)) { try { const csvUrl = `https://docs.google.com/spreadsheets/d/${sheetId}/gviz/tq?tqx=out:csv&sheet=${encodeURIComponent(tabName)}`; const response = await fetch(csvUrl); const text = await response.text(); // Check if Google returned HTML (private sheet / login redirect) instead of CSV if (text.trim().startsWith('Sign in')) { console.warn(`Tab "${tabName}" returned HTML/Auth page. Ensure sheet sharing is set to 'Anyone with the link can view'.`); cachedSheetData[key] = []; continue; } Papa.parse(text, { header: true, skipEmptyLines: true, complete: function(res) { cachedSheetData[key] = (res.data || []).filter(r => Object.values(r).some(v => v !== null && v !== '')); } }); } catch (e) { console.warn(`Tab ${tabName} fetch warning`, e); cachedSheetData[key] = []; } } toast.className = 'mb-6 p-4 rounded-xl text-xs font-medium flex items-center justify-between glass-card border-emerald-500/40 text-emerald-300'; toast.innerHTML = ` Telemetry sync evaluated across tabs! Check sharing permissions if preview is empty.`; setTimeout(() => toast.classList.add('hidden'), 5000); renderPageData(currentPage); }
